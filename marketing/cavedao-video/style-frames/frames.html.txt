    <script>
      // The gate mark, rasterised onto its own 4-unit grid (16 × 16 cells).
      const GATE = [[4,60],[4,28],[8,28],[8,20],[12,20],[12,16],[16,16],[16,12],[24,12],[24,8],[40,8],[40,12],[48,12],[48,16],[52,16],[52,20],[56,20],[56,28],[60,28],[60,60],[44,60],[44,32],[40,32],[40,24],[36,24],[36,20],[28,20],[28,24],[24,24],[24,32],[20,32],[20,60]];
      function inside(x, y) {
        let c = false;
        for (let i = 0, j = GATE.length - 1; i < GATE.length; j = i++) {
          const [xi, yi] = GATE[i], [xj, yj] = GATE[j];
          if ((yi > y) !== (yj > y) && x < ((xj - xi) * (y - yi)) / (yj - yi) + xi) c = !c;
        }
        return c;
      }
      const CELLS = [];
      for (let r = 0; r < 16; r++) for (let q = 0; q < 16; q++) if (inside(q * 4 + 2, r * 4 + 2)) CELLS.push([q, r]);
      // Bottom rows first, as if stacked from the floor up.
      CELLS.sort((m, n) => n[1] - m[1] || Math.abs(m[0] - 7.5) - Math.abs(n[0] - 7.5));
      const gateA = document.getElementById("gateA");
      CELLS.forEach(([q, r], k) => {
        const i = document.createElement("i");
        i.style.left = q * 28 + "px";
        i.style.top = r * 28 + "px";
        // Take A shows the build mid-way: the top blocks still dropping in.
        const left = k / CELLS.length;
        if (left > 0.78) { i.style.transform = `translateY(${-Math.round((left - 0.78) * 900 / 28) * 28}px)`; i.style.opacity = 0.9; }
        gateA.appendChild(i);
      });

      const wall = document.getElementById("wall");
      for (let k = 0; k < 40; k++) {
        const img = document.createElement("img");
        img.src = `assets/art/item-${String(k % 36).padStart(2, "0")}.png`;
        wall.appendChild(img);
      }

      const tl = gsap.timeline({ paused: true });
      tl.set({}, {}, 3);
      window.__timelines = window.__timelines || {};
      window.__timelines["frames"] = tl;
    </script>
